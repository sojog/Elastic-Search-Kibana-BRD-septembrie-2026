# 1. Gășiți articole despre Donald Trump în contextul NATO
GET news/_search
{
  "query": {
    "bool": {
      "must": [
        {
          "multi_match": {
            "query": "Donald Trump",
            "fields": ["headline", "short_description"]
          }
        },
        {
            "multi_match": {
            "query": "NATO",
            "fields": ["headline", "short_description"]
          }}
      ]
    }
  }
}

# Pentru a nu intoarce intreaga lista de "hits"
GET news/_search
{
  "size": 0,
  "query": {
    "bool": {
      "must": [
        {
          "multi_match": {
            "query": "Donald Trump",
            "fields": ["headline", "short_description"]
          }
        },
        {
            "multi_match": {
            "query": "NATO",
            "fields": ["headline", "short_description"]
          }}
      ]
    }
  }
}


# 2. Găsiți articole despre Donald Trump scrise de Lee Moran

{

                {
                    "multi_match": {

                    }
                },
                {"match_phrase": {
                  "authors": "Lee Moran"
                }}
    
         ]
    }
  }
}

# 3. Găsiți toate articolele despre Donald Trump înainte ca el să fie președinte
GET news/_search  
{ "query":{
    "bool": {"must": [
         {  "multi_match": {
                    "query": "Donald Trump",
                    "fields": ["headline", "short_description"]
                    }}
                    
    ],
    "filter": [
      {
        "range": {
        "date": {
          "lte": "2017-01-01"
        }
      }}
    ]}
}
}


# 4. Găsiți toate articolele despre Donald Trump care nu sunt în sfera politică.
GET news/_search  
{ 
    "query":{
    "bool": {
        "must": [
         {  "multi_match": {
                    "query": "Donald Trump",
                    "fields": ["headline", "short_description"]
                    }}
                    
        ],
    "must_not": [
      {
        "term": {
          "category": "POLITICS"
        }
      }
    ]
    }
    }
}

# 5. Toate articolele despre Michelle Obama, dar nu îl includ pe Barack

                    "query": "Michelle Obama",

                    }}
                    
        ],
        "must_not": [
           {
            "multi_match": {
              "query": "Barack",

            }
           }
        ]
    }
    }
}
# 6. Toate articolele care conțin cel puțin 2 surori Kardashian menționate
GET news/_search
{
  "query": {
    "bool": {
      "should": [
        { "match": { "headline": "Kim Kardashian" } },
        { "match": { "headline": "Kourtney Kardashian" } },
        { "match": { "headline": "Khloe Kardashian" } },
        { "match": { "headline": "Kendall Jenner" } },
        { "match": { "headline": "Kylie Jenner" } }
      ],
      "minimum_should_match": 2
    }
  }
}
